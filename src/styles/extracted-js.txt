    let currentLang = localStorage.getItem('lang') || 'zh';

    function setLang(lang) {
      currentLang = lang;
      localStorage.setItem('lang', lang);
      applyTranslations();
      renderPortfolio(); // re-render cards with new lang
      if (window._heroDescTyper && window.heroDescTexts) window._heroDescTyper.restart(window.heroDescTexts[lang]);
    }

    function updateSmLangBtns() {
      const zh = document.getElementById('sm-lang-zh');
      const en = document.getElementById('sm-lang-en');
      if (!zh || !en) return;
      zh.classList.toggle('active', currentLang === 'zh');
      en.classList.toggle('active', currentLang === 'en');
    }

    function toggleLangDropdown(e) {
      if (e) e.stopPropagation();
      const dd = document.getElementById('lang-dropdown');
      const chevron = document.getElementById('lang-chevron');
      const isOpen = dd.classList.contains('open');
      dd.classList.toggle('open');
      if (chevron) chevron.style.transform = isOpen ? '' : 'rotate(180deg)';
    }
    function closeLangDropdown() {
      const dd = document.getElementById('lang-dropdown');
      const chevron = document.getElementById('lang-chevron');
      dd.classList.remove('open');
      if (chevron) chevron.style.transform = '';
    }
    document.addEventListener('click', function (e) {
      if (!e.target.closest('#lang-menu-btn') && !e.target.closest('#lang-dropdown')) {
        closeLangDropdown();
      }
    });

    function applyTranslations() {
      const t = translations[currentLang];
      document.querySelectorAll('[data-i18n]').forEach(el => {
        const key = el.getAttribute('data-i18n');
        if (t[key] !== undefined) el.innerHTML = t[key];
      });
      // update globe button label
      const langLabel = document.getElementById('lang-label');
      if (langLabel) langLabel.textContent = currentLang === 'zh' ? '繁體中文' : 'English';
      // update checkmarks
      document.querySelectorAll('.lang-opt').forEach(btn => btn.classList.remove('active'));
      const activeOpt = document.querySelector(`.lang-opt[onclick*="'${currentLang}'"]`);
      if (activeOpt) activeOpt.classList.add('active');
    }

    function makeCard(p, mode) {
      const t = translations[currentLang];
      const title = t[p.id + '_title'] || p.id;
      const desc = t[p.id + '_desc'] || '';
      const tags = p.tags.map(tag => `<span class="project-tag">${tag}</span>`).join('');
      const cls = mode === 'grid' ? 'grid-card' : 'project-card';
      return `<a class="${cls}" href="${p.link}" data-category="${p.category}">
      <img src="${p.img}" alt="${title}" loading="lazy">
      <div class="project-overlay">
        <div class="project-title">${title}</div>
        <div class="project-desc">${desc}</div>
        <div class="project-tags">${tags}</div>
      </div>
    </a>`;
    }

    function renderPortfolio() {
      let desk1HTML = '', desk2HTML = '', gridHTML = '';

      PROJECTS.forEach((p, i) => {
        const scrollCard = makeCard(p, 'scroll');
        const gridCard = makeCard(p, 'grid');

        if (i < 7) desk1HTML += scrollCard;
        else desk2HTML += scrollCard;

        gridHTML += gridCard;
      });

      const t1 = document.getElementById('track-desk-1');
      const t2 = document.getElementById('track-desk-2');
      if (t1) t1.innerHTML = desk1HTML;
      if (t2) t2.innerHTML = desk2HTML;

      const grid = document.getElementById('portfolio-grid');
      if (grid) {
        grid.innerHTML = gridHTML;
        /* apply bento big-card spans */
        const bigIdx = getBentoBigIndices(PROJECTS.length);
        grid.querySelectorAll('.grid-card').forEach((c, i) => {
          if (bigIdx.includes(i)) c.classList.add('mb-big');
        });
        /* re-apply MagicBento if the grid is currently expanded */
        if (grid.style.display !== 'none') initMagicBento();
      }

      // reset & re-init portfolio scrollers (CSS animation)
      document.querySelectorAll('.portfolio-scroller').forEach(s => {
        // remove old clones
        s.querySelectorAll('[aria-hidden="true"]').forEach(el => el.remove());
        s.removeAttribute('data-animated');
        s.removeAttribute('data-raf-init');

        const inner = s.querySelector('.scroller-inner');
        if (!inner) return;

        // clone items so total = 2× (CSS translate(-50%) needs this)
        const origItems = Array.from(inner.children);
        origItems.forEach(item => {
          const clone = item.cloneNode(true);
          clone.setAttribute('aria-hidden', 'true');
          inner.appendChild(clone);
        });
        s.setAttribute('data-animated', 'true');
      });
    }

    /* ── BENTO GRID: random big-card indices (stable per session) ── */
    let _bentoBigIndices = null;
    function getBentoBigIndices(total) {
      if (!_bentoBigIndices) {
        // Pick one from each half so big cards are distributed across the grid
        const a = Math.floor(Math.random() * Math.floor(total / 2));
        const b = Math.floor(total / 2) + Math.floor(Math.random() * Math.ceil(total / 2));
        _bentoBigIndices = [a, b];
      }
      return _bentoBigIndices;
    }

    /* ── MAGIC BENTO: spotlight + border-glow + tilt + magnetism ── */
    let _mbSpotlight = null;
    let _mbMoveHandler = null;
    let _mbLeaveHandler = null;
    let _mbGridLeaveHandler = null;
    let _mbCurrentCard = null;

    const MB_GLOW = '74, 0, 224';
    const MB_RADIUS = 170;

    function destroyMagicBento() {
      if (_mbSpotlight) { _mbSpotlight.remove(); _mbSpotlight = null; }
      if (_mbMoveHandler) { document.removeEventListener('mousemove', _mbMoveHandler); _mbMoveHandler = null; }
      if (_mbLeaveHandler) { document.removeEventListener('mouseleave', _mbLeaveHandler); _mbLeaveHandler = null; }
      const g = document.getElementById('portfolio-grid');
      if (g && _mbGridLeaveHandler) g.removeEventListener('mouseleave', _mbGridLeaveHandler);
      _mbGridLeaveHandler = null;
      _mbCurrentCard = null;
    }

    function initMagicBento() {
      destroyMagicBento();
      const grid = document.getElementById('portfolio-grid');
      if (!grid || grid.style.display === 'none') return;

      /* add glow class + transform-origin to each card */
      grid.querySelectorAll('.grid-card').forEach(c => {
        c.classList.add('mb-glow');
        c.style.transformOrigin = '50% 50%';
      });

      /* spotlight element */
      const sl = document.createElement('div');
      sl.className = 'mb-spotlight';
      document.body.appendChild(sl);
      _mbSpotlight = sl;

      const proximity = MB_RADIUS * 0.5;
      const fadeDistance = MB_RADIUS * 0.75;

      function updateGlow(card, mx, my, intensity) {
        const r = card.getBoundingClientRect();
        card.style.setProperty('--glow-x', `${((mx - r.left) / r.width * 100).toFixed(1)}%`);
        card.style.setProperty('--glow-y', `${((my - r.top) / r.height * 100).toFixed(1)}%`);
        card.style.setProperty('--glow-intensity', intensity.toFixed(3));
        card.style.setProperty('--glow-radius', `${MB_RADIUS}px`);
      }

      function resetCard(card) {
        gsap.to(card, { rotateX: 0, rotateY: 0, x: 0, y: 0, duration: 0.4, ease: 'power3.out', overwrite: 'auto' });
        card.style.setProperty('--glow-intensity', '0');
      }

      _mbMoveHandler = e => {
        const section = document.getElementById('portfolio');
        if (!section) return;
        const sr = section.getBoundingClientRect();
        const inSection = e.clientX >= sr.left && e.clientX <= sr.right &&
          e.clientY >= sr.top && e.clientY <= sr.bottom;

        const cards = grid.querySelectorAll('.grid-card.mb-glow');

        if (!inSection || grid.style.display === 'none') {
          gsap.to(sl, { opacity: 0, duration: 0.3, ease: 'power2.out' });
          cards.forEach(c => c.style.setProperty('--glow-intensity', '0'));
          return;
        }

        /* move spotlight */
        gsap.to(sl, { left: e.clientX, top: e.clientY, duration: 0.12, ease: 'power2.out' });

        /* per-card border-glow + spotlight opacity */
        let minDist = Infinity;
        cards.forEach(card => {
          const r = card.getBoundingClientRect();
          const cx = r.left + r.width / 2;
          const cy = r.top + r.height / 2;
          const d = Math.max(0, Math.hypot(e.clientX - cx, e.clientY - cy) - Math.max(r.width, r.height) / 2);
          minDist = Math.min(minDist, d);
          const gi = d <= proximity ? 1 :
            d <= fadeDistance ? (fadeDistance - d) / (fadeDistance - proximity) : 0;
          updateGlow(card, e.clientX, e.clientY, gi);
        });

        const targetOp = minDist <= proximity ? 0.8 :
          minDist <= fadeDistance ? ((fadeDistance - minDist) / (fadeDistance - proximity)) * 0.8 : 0;
        gsap.to(sl, { opacity: targetOp, duration: targetOp > 0 ? 0.2 : 0.5, ease: 'power2.out' });

        /* tilt + magnetism on hovered card */
        const hovered = e.target.closest?.('.grid-card.mb-glow');
        if (_mbCurrentCard && _mbCurrentCard !== hovered) resetCard(_mbCurrentCard);
        if (hovered) {
          _mbCurrentCard = hovered;
          const r = hovered.getBoundingClientRect();
          const lx = e.clientX - r.left;
          const ly = e.clientY - r.top;
          const cx = r.width / 2;
          const cy = r.height / 2;
          gsap.to(hovered, {
            rotateX: ((ly - cy) / cy) * -9,
            rotateY: ((lx - cx) / cx) * 9,
            x: (lx - cx) * 0.04,
            y: (ly - cy) * 0.04,
            transformPerspective: 900,
            duration: 0.15,
            ease: 'power2.out',
            overwrite: 'auto',
          });
        }
      };

      _mbLeaveHandler = () => {
        gsap.to(sl, { opacity: 0, duration: 0.35, ease: 'power2.out' });
        grid.querySelectorAll('.grid-card.mb-glow').forEach(resetCard);
        _mbCurrentCard = null;
      };

      _mbGridLeaveHandler = e => {
        if (!e.relatedTarget || !grid.contains(e.relatedTarget)) {
          if (_mbCurrentCard) { resetCard(_mbCurrentCard); _mbCurrentCard = null; }
        }
      };

      document.addEventListener('mousemove', _mbMoveHandler);
      document.addEventListener('mouseleave', _mbLeaveHandler);
      grid.addEventListener('mouseleave', _mbGridLeaveHandler);
    }

    function initProfileCard() {
      var wrap = document.getElementById('profile-card-wrap');
      var shell = document.getElementById('profile-card-shell');
      var card = document.getElementById('profile-card-el');
      if (!wrap || !shell || !card) return;

      // ── spring tilt engine ──
      var rafId = null, running = false, lastTs = 0;
      var curX = 0, curY = 0, tgtX = 0, tgtY = 0;
      var DEFAULT_TAU = 0.14, INITIAL_TAU = 0.6, initialUntil = 0;
      var leaveRaf = null, enterTimer = null;

      function clampN(v, lo, hi) { return Math.min(Math.max(v, lo), hi); }
      function adj(v, f0, f1, t0, t1) { return parseFloat((t0 + (t1 - t0) * (v - f0) / (f1 - f0)).toFixed(3)); }

      function applyXY(x, y) {
        var w = shell.clientWidth || 1;
        var h = shell.clientHeight || 1;
        var px = clampN((100 / w) * x, 0, 100);
        var py = clampN((100 / h) * y, 0, 100);
        var cx = px - 50, cy = py - 50;
        wrap.style.setProperty('--pointer-x', px + '%');
        wrap.style.setProperty('--pointer-y', py + '%');
        wrap.style.setProperty('--background-x', adj(px, 0, 100, 35, 65) + '%');
        wrap.style.setProperty('--background-y', adj(py, 0, 100, 35, 65) + '%');
        wrap.style.setProperty('--pointer-from-center', clampN(Math.hypot(py - 50, px - 50) / 50, 0, 1));
        wrap.style.setProperty('--pointer-from-top', py / 100);
        wrap.style.setProperty('--pointer-from-left', px / 100);
        wrap.style.setProperty('--rotate-x', parseFloat((-(cx / 5)).toFixed(3)) + 'deg');
        wrap.style.setProperty('--rotate-y', parseFloat((cy / 4).toFixed(3)) + 'deg');
      }

      function step(ts) {
        if (!running) return;
        if (!lastTs) lastTs = ts;
        var dt = (ts - lastTs) / 1000; lastTs = ts;
        var tau = ts < initialUntil ? INITIAL_TAU : DEFAULT_TAU;
        var k = 1 - Math.exp(-dt / tau);
        curX += (tgtX - curX) * k; curY += (tgtY - curY) * k;
        applyXY(curX, curY);
        if (Math.abs(tgtX - curX) > 0.05 || Math.abs(tgtY - curY) > 0.05 || document.hasFocus()) {
          rafId = requestAnimationFrame(step);
        } else { running = false; lastTs = 0; if (rafId) { cancelAnimationFrame(rafId); rafId = null; } }
      }

      function startRaf() { if (running) return; running = true; lastTs = 0; rafId = requestAnimationFrame(step); }
      function setTarget(x, y) { tgtX = x; tgtY = y; startRaf(); }
      function toCenter() { setTarget(shell.clientWidth / 2, shell.clientHeight / 2); }
      function getOff(e) { var r = shell.getBoundingClientRect(); return { x: e.clientX - r.left, y: e.clientY - r.top }; }

      shell.addEventListener('pointerenter', function (e) {
        card.classList.add('active');
        shell.classList.add('entering');
        if (enterTimer) clearTimeout(enterTimer);
        enterTimer = setTimeout(function () { shell.classList.remove('entering'); }, 180);
        var o = getOff(e); setTarget(o.x, o.y);
      });
      shell.addEventListener('pointermove', function (e) { var o = getOff(e); setTarget(o.x, o.y); });
      shell.addEventListener('pointerleave', function () {
        toCenter();
        function checkSettle() {
          if (Math.hypot(tgtX - curX, tgtY - curY) < 0.6) { card.classList.remove('active'); leaveRaf = null; }
          else { leaveRaf = requestAnimationFrame(checkSettle); }
        }
        if (leaveRaf) cancelAnimationFrame(leaveRaf);
        leaveRaf = requestAnimationFrame(checkSettle);
      });

      // Initial entrance animation: start from top-right, ease to center
      curX = (shell.clientWidth || 0) - 70; curY = 60;
      applyXY(curX, curY);
      toCenter();
      initialUntil = performance.now() + 1200;
      startRaf();
    }

    function animateGridIn(cards) {
      if (!cards || !cards.length) return;
      // Kill any in-flight tweens so their clearProps won't remove display:none
      // set by the filter on hidden cards.
      gsap.killTweensOf('#portfolio-grid .grid-card');
      const arr = Array.from(cards);
      arr.forEach(c => { c.style.transition = 'none'; });
      gsap.fromTo(arr,
        { y: 60, opacity: 0, filter: 'blur(8px)', scale: 0.95 },
        { y: 0, opacity: 1, filter: 'blur(0px)', scale: 1,
          duration: 0.65, ease: 'power3.out', stagger: 0.045,
          clearProps: 'transform,opacity,filter',
          onComplete() { arr.forEach(c => { c.style.transition = ''; }); }
        }
      );
    }

    function initPortfolioToggle() {
      const btn = document.getElementById('toggle-portfolio-view');
      if (!btn) return;
      const btnText = document.getElementById('toggle-portfolio-text');
      const btnIcon = document.getElementById('toggle-portfolio-icon');
      const scrollDesk = document.getElementById('portfolio-scroller-desktop');
      const grid = document.getElementById('portfolio-grid');
      const filters = document.getElementById('portfolio-filters');

      // inject filter buttons
      filters.innerHTML = `
      <button class="btn-glass filter-btn active" data-filter="all">All</button>
      <button class="btn-glass filter-btn" data-filter="mobile">Apps Design</button>
      <button class="btn-glass filter-btn" data-filter="web">Web Design</button>`;

      let expanded = false;

      function doExpand(savedFilter) {
        expanded = true;
        scrollDesk.style.display = 'none';

        // Pre-hide all cards before making the grid visible so there is no
        // flash of un-animated content during the scroll-settle delay.
        if (savedFilter === null) {
          document.querySelectorAll('#portfolio-grid .grid-card').forEach(c => {
            c.style.opacity = '0';
          });
        }

        grid.style.display = 'grid';
        filters.style.display = 'flex';
        initMagicBento();
        btnText.innerText = 'Fold';
        btnIcon.className = 'fas fa-chevron-up';
        sessionStorage.setItem('portfolioExpanded', 'true');

        // restore or reset filter
        const f = savedFilter || 'all';
        filters.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
        const target = filters.querySelector(`[data-filter="${f}"]`);
        if (target) target.classList.add('active');
        const visibleCards = [];
        document.querySelectorAll('#portfolio-grid .grid-card').forEach(c => {
          const show = (f === 'all' || c.dataset.category === f);
          c.style.display = show ? '' : 'none';
          if (show) visibleCards.push(c);
        });
        sessionStorage.setItem('portfolioActiveFilter', f);

        // animate only on fresh expand, not on page-restore (delay past the 50ms scroll jump)
        if (savedFilter === null) setTimeout(() => animateGridIn(visibleCards), 120);
      }

      function doFold() {
        expanded = false;
        destroyMagicBento();
        scrollDesk.style.display = '';
        grid.style.display = 'none';
        filters.style.display = 'none';
        btnText.innerText = 'View All Projects';
        btnIcon.className = 'fas fa-arrow-right';
        sessionStorage.setItem('portfolioExpanded', 'false');

        // scroll back to portfolio top
        const portfolioSec = document.getElementById('portfolio');
        if (portfolioSec) {
          const header = document.getElementById('main-header');
          const headerOffset = header ? header.offsetHeight + 20 : 80;
          const top = portfolioSec.getBoundingClientRect().top + window.pageYOffset - headerOffset;
          window.scrollTo({ top, behavior: 'smooth' });
        }
      }

      btn.addEventListener('click', function () {
        if (expanded) {
          doFold();
        } else {
          const portfolioSec = document.getElementById('portfolio');
          const header = document.getElementById('main-header');
          const headerOffset = header ? header.offsetHeight + 20 : 80;
          const targetScrollY = portfolioSec
            ? portfolioSec.getBoundingClientRect().top + window.pageYOffset - headerOffset
            : 0;

          // Disable scroll anchoring so the browser doesn't auto-scroll when the
          // grid is inserted (scroll anchoring would push the viewport down by the
          // height of the newly inserted content, fighting our intended scroll).
          document.documentElement.style.overflowAnchor = 'none';

          doExpand(null);

          // After layout settles, jump instantly to portfolio top then re-enable anchoring
          setTimeout(() => {
            window.scrollTo({ top: targetScrollY, behavior: 'instant' });
            setTimeout(() => {
              document.documentElement.style.overflowAnchor = '';
            }, 100);
          }, 50);
        }
      });

      // ── RESTORE STATE on page load ──
      const savedExpanded = sessionStorage.getItem('portfolioExpanded');
      const savedFilter = sessionStorage.getItem('portfolioActiveFilter');
      const savedScrollY = sessionStorage.getItem('portfolioScrollY');

      if (savedExpanded === 'true') {
        doExpand(savedFilter);
      }

      // Always restore scroll when returning from a project page,
      // regardless of expanded state. Clear after use so reload stays at top.
      if (savedScrollY !== null) {
        sessionStorage.removeItem('portfolioScrollY');
        requestAnimationFrame(() => {
          requestAnimationFrame(() => {
            window.scrollTo({ top: parseFloat(savedScrollY), behavior: 'instant' });
          });
        });
      }
    }

    (function initNavCircles() {
      var navLinks = document.querySelectorAll('.nav-link');
      var tlMap = new Map();
      var tweenMap = new Map();

      navLinks.forEach(function (link) {
        var circle = link.querySelector('.nav-circle');
        var label = link.querySelector('.nav-label');
        var labelHover = link.querySelector('.nav-label-hover');
        if (!circle || !label || !labelHover) return;

        function layout() {
          var rect = link.getBoundingClientRect();
          var w = rect.width, h = rect.height;
          if (!w || !h) return;
          var R = ((w * w) / 4 + h * h) / (2 * h);
          var D = Math.ceil(2 * R) + 2;
          var delta = Math.ceil(R - Math.sqrt(Math.max(0, R * R - (w * w) / 4))) + 1;
          var originY = D - delta;
          circle.style.width = D + 'px';
          circle.style.height = D + 'px';
          circle.style.bottom = -delta + 'px';
          gsap.set(circle, { xPercent: -50, scale: 0, transformOrigin: '50% ' + originY + 'px' });
          gsap.set(label, { y: 0 });
          gsap.set(labelHover, { y: Math.ceil(h + 12), opacity: 0 });
          var old = tlMap.get(link);
          if (old) old.kill();
          var tl = gsap.timeline({ paused: true });
          tl.to(circle, { scale: 1.2, xPercent: -50, duration: 2, ease: 'power3.out' }, 0);
          tl.to(label, { y: -(h + 8), duration: 2, ease: 'power3.out' }, 0);
          tl.to(labelHover, { y: 0, opacity: 1, duration: 2, ease: 'power3.out' }, 0);
          tlMap.set(link, tl);
        }

        layout();
        window.addEventListener('resize', layout);
        window.addEventListener('load', layout, { once: true });

        link.addEventListener('mouseenter', function () {
          var tl = tlMap.get(link);
          if (!tl) return;
          tweenMap.get(link) && tweenMap.get(link).kill();
          tweenMap.set(link, gsap.to(tl, { time: tl.duration(), duration: 0.3, ease: 'power3.out', overwrite: 'auto' }));
        });

        link.addEventListener('mouseleave', function () {
          var tl = tlMap.get(link);
          if (!tl) return;
          tweenMap.get(link) && tweenMap.get(link).kill();
          tweenMap.set(link, gsap.to(tl, { time: 0, duration: 0.2, ease: 'power3.out', overwrite: 'auto' }));
        });
      });
    })();

    // ── NAV HIGHLIGHT ON SCROLL ──
    const sections = document.querySelectorAll('section[id]');

    // ── RAF SCROLLER ENGINE (LogoLoop-inspired) ──
    // No CSS animation — position driven by requestAnimationFrame.
    // Smooth lerp on hover: velocity eases to hoverSpeed, zero jump.

    const SMOOTH_TAU = 0.18; // lerp time constant (seconds)

    function createScroller(scroller, normalSpeed, hoverSpeed) {
      const inner = scroller.querySelector('.scroller-inner');
      if (!inner) return;
      const isReverse = scroller.getAttribute('data-direction') === 'right';

      if (scroller.getAttribute('data-raf-init') === 'true') return;

      // Snapshot original items BEFORE cloning (HTMLCollection is live)
      const origItems = Array.from(inner.children);
      origItems.forEach(item => {
        const clone = item.cloneNode(true);
        clone.setAttribute('aria-hidden', 'true');
        inner.appendChild(clone);
      });
      scroller.setAttribute('data-raf-init', 'true');

      // Measure one-set width — wait for images to load first
      const gap = parseFloat(getComputedStyle(inner).gap) || 10;

      function startRAF() {
        const oneSetWidth = origItems.reduce((sum, el) => sum + el.offsetWidth + gap, 0);
        if (oneSetWidth <= 0) return;

        let offset = 0;
        let velocity = normalSpeed;
        let targetVelocity = normalSpeed;
        let lastTs = null;

        const direction = isReverse ? -1 : 1;

        function tick(ts) {
          if (!lastTs) lastTs = ts;
          const dt = Math.min((ts - lastTs) / 1000, 0.05);
          lastTs = ts;
          const factor = 1 - Math.exp(-dt / SMOOTH_TAU);
          velocity += (targetVelocity - velocity) * factor;
          offset += velocity * dt * direction;
          offset = ((offset % oneSetWidth) + oneSetWidth) % oneSetWidth;
          inner.style.transform = `translateX(${-offset}px)`;
          requestAnimationFrame(tick);
        }

        requestAnimationFrame(tick);

        scroller.addEventListener('mouseenter', () => { targetVelocity = hoverSpeed; });
        scroller.addEventListener('mouseleave', () => { targetVelocity = normalSpeed; });
      }

      // Wait for images (skills row 1 has img logos)
      const images = inner.querySelectorAll('img');
      if (images.length === 0) {
        startRAF();
      } else {
        let loaded = 0;
        const onLoad = () => { if (++loaded >= images.length) startRAF(); };
        images.forEach(img => {
          if (img.complete) onLoad();
          else { img.addEventListener('load', onLoad, { once: true }); img.addEventListener('error', onLoad, { once: true }); }
        });
      }
    }

    function initInfiniteScroll() {
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

      // Hero tags — items already pre-cloned in HTML (2 sets)
      // Just start RAF directly, don't clone again
      document.querySelectorAll('.hero-scroller').forEach(scroller => {
        const inner = scroller.querySelector('.scroller-inner');
        if (!inner || scroller.getAttribute('data-raf-init') === 'true') return;
        scroller.setAttribute('data-raf-init', 'true');

        // oneSetWidth = half of total (since HTML has 2 sets)
        const gap = parseFloat(getComputedStyle(inner).gap) || 12;
        const allItems = Array.from(inner.children);
        const halfCount = allItems.length / 2;
        const oneSetWidth = allItems.slice(0, halfCount).reduce((sum, el) => sum + el.offsetWidth + gap, 0);
        if (oneSetWidth <= 0) return;

        let offset = 0, velocity = 80, targetVelocity = 80, lastTs = null;

        function tick(ts) {
          if (!lastTs) lastTs = ts;
          const dt = Math.min((ts - lastTs) / 1000, 0.05);
          lastTs = ts;
          velocity += (targetVelocity - velocity) * (1 - Math.exp(-dt / SMOOTH_TAU));
          offset = ((offset + velocity * dt) % oneSetWidth + oneSetWidth) % oneSetWidth;
          inner.style.transform = `translateX(${-offset}px)`;
          requestAnimationFrame(tick);
        }
        requestAnimationFrame(tick);
        scroller.addEventListener('mouseenter', () => { targetVelocity = 25; });
        scroller.addEventListener('mouseleave', () => { targetVelocity = 80; });
      });

      // Skills scrollers — createScroller handles cloning
      document.querySelectorAll('.skills-scroller').forEach(s => createScroller(s, 80, 25));
    }

    // ── BORDER GLOW — My Design Process cards ──
    function initBorderGlowCards() {
      const GLOW_COLOR = '264 70 75';           // HSL: purple matching --primary
      const BG = '#13101c';
      const COLORS = ['#6C63FF', '#f472b6', '#38bdf8'];
      const INTENSITY = 1.1;
      const GRAD_POS = ['80% 55%', '69% 34%', '8% 6%', '41% 38%', '86% 85%', '82% 18%', '51% 4%'];
      const GRAD_KEYS = ['--gradient-one', '--gradient-two', '--gradient-three', '--gradient-four', '--gradient-five', '--gradient-six', '--gradient-seven'];
      const COLOR_MAP = [0, 1, 2, 0, 1, 2, 1];

      function parseHSL(s) {
        const m = s.match(/([\d.]+)\s*([\d.]+)%?\s*([\d.]+)%?/);
        return m ? { h: +m[1], s: +m[2], l: +m[3] } : { h: 264, s: 70, l: 75 };
      }
      function setGlowVars(el, hslStr, intensity) {
        const { h, s, l } = parseHSL(hslStr);
        const base = `${h}deg ${s}% ${l}%`;
        [100, 60, 50, 40, 30, 20, 10].forEach((op, i) => {
          const suffix = i === 0 ? '' : `-${op}`;
          el.style.setProperty(`--glow-color${suffix}`, `hsl(${base} / ${Math.min(op * intensity, 100)}%)`);
        });
      }
      function setGradientVars(el, colors) {
        GRAD_KEYS.forEach((key, i) => {
          const c = colors[Math.min(COLOR_MAP[i], colors.length - 1)];
          el.style.setProperty(key, `radial-gradient(at ${GRAD_POS[i]}, ${c} 0px, transparent 50%)`);
        });
        el.style.setProperty('--gradient-base', `linear-gradient(${colors[0]} 0 100%)`);
      }
      function getCenter(el) {
        const r = el.getBoundingClientRect();
        return [r.width / 2, r.height / 2];
      }
      function edgeProximity(el, x, y) {
        const [cx, cy] = getCenter(el);
        const dx = x - cx, dy = y - cy;
        let kx = Infinity, ky = Infinity;
        if (dx !== 0) kx = cx / Math.abs(dx);
        if (dy !== 0) ky = cy / Math.abs(dy);
        return Math.min(Math.max(1 / Math.min(kx, ky), 0), 1);
      }
      function cursorAngle(el, x, y) {
        const [cx, cy] = getCenter(el);
        const dx = x - cx, dy = y - cy;
        if (!dx && !dy) return 0;
        let deg = Math.atan2(dy, dx) * (180 / Math.PI) + 90;
        return deg < 0 ? deg + 360 : deg;
      }

      document.querySelectorAll('.process-card').forEach(card => {
        // Inject edge-light span
        const el = document.createElement('span');
        el.className = 'edge-light';
        card.insertBefore(el, card.firstChild);

        // Wrap existing content in border-glow-inner
        const inner = document.createElement('div');
        inner.className = 'border-glow-inner';
        Array.from(card.children)
          .filter(c => !c.classList.contains('edge-light'))
          .forEach(c => inner.appendChild(c));
        card.appendChild(inner);

        // Apply class + CSS vars
        card.classList.add('border-glow-card');
        card.style.setProperty('--card-bg', BG);
        setGlowVars(card, GLOW_COLOR, INTENSITY);
        setGradientVars(card, COLORS);

        // Pointer tracking
        card.addEventListener('pointermove', e => {
          const rect = card.getBoundingClientRect();
          const x = e.clientX - rect.left;
          const y = e.clientY - rect.top;
          card.style.setProperty('--edge-proximity', (edgeProximity(card, x, y) * 100).toFixed(3));
          card.style.setProperty('--cursor-angle', `${cursorAngle(card, x, y).toFixed(3)}deg`);
        });
      });
    }

    // ── SCROLL ANIMATIONS — section-aware top-to-bottom cascade ──
    function initScrollAnimations() {
      const ROW_BAND = 80;   // px — elements within this vertical band are treated as the same row
      const ROW_DELAY = 230;  // ms between each row
      const COL_DELAY = 110;  // ms between sibling elements within the same row

      function revealSection(section) {
        const items = Array.from(section.querySelectorAll('.stagger-item, .fade-in'))
          .filter(el => !el.classList.contains('visible'));
        if (!items.length) return;

        // Positions relative to section top so row keys start at 0
        const secTop = section.getBoundingClientRect().top;
        const measured = items.map(el => {
          const r = el.getBoundingClientRect();
          return { el, relTop: r.top - secTop, left: r.left };
        }).sort((a, b) => {
          const ra = Math.round(a.relTop / ROW_BAND);
          const rb = Math.round(b.relTop / ROW_BAND);
          return ra !== rb ? ra - rb : a.left - b.left;
        });

        let baseRowKey = -1, rowIdx = -1, colIdx = 0;
        measured.forEach(({ el, relTop }) => {
          const key = Math.round(relTop / ROW_BAND);
          if (key !== baseRowKey) { baseRowKey = key; rowIdx++; colIdx = 0; }
          else colIdx++;
          const delay = rowIdx * ROW_DELAY + colIdx * COL_DELAY;
          setTimeout(() => el.classList.add('visible'), delay);
        });
      }

      // Hero: reveal immediately on load (GSAP handles h1/h2 via bypass)
      revealSection(document.getElementById('home'));

      // All other sections: reveal when they enter the viewport
      const seen = new Set();
      const sections = Array.from(document.querySelectorAll('section.section'));

      function checkSections() {
        sections.forEach(sec => {
          if (seen.has(sec)) return;
          const r = sec.getBoundingClientRect();
          if (r.top < window.innerHeight * 0.92 && r.bottom > 0) {
            seen.add(sec);
            revealSection(sec);
          }
        });
      }

      // IntersectionObserver as primary trigger
      const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (!entry.isIntersecting) return;
          const sec = entry.target;
          if (seen.has(sec)) return;
          seen.add(sec);
          observer.unobserve(sec);
          revealSection(sec);
        });
      }, { threshold: 0.05 });
      sections.forEach(sec => observer.observe(sec));

      // Scroll listener as reliable fallback
      window.addEventListener('scroll', checkSections, { passive: true });
      checkSections();
    }

    // ── HERO SPLIT TEXT ──
    function initHeroSplitText() {
      var EASE = 'power3.out';

      // Split a single element's child nodes into word spans, preserving inner elements
      function splitWords(el) {
        var result = [];
        var nodes = Array.from(el.childNodes);
        while (el.firstChild) el.removeChild(el.firstChild);
        nodes.forEach(function (n) {
          if (n.nodeType === 3) {
            n.textContent.split(/(\s+)/).forEach(function (part) {
              if (/^\s+$/.test(part)) { el.appendChild(document.createTextNode(part)); }
              else if (part) {
                var s = document.createElement('span');
                s.style.cssText = 'display:inline-block;will-change:transform,opacity';
                s.textContent = part;
                el.appendChild(s); result.push(s);
              }
            });
          } else if (n.nodeType === 1) {
            if (n.nodeName === 'BR') { el.appendChild(n); return; }
            var s = document.createElement('span');
            s.style.cssText = 'display:inline-block;will-change:transform,opacity';
            s.appendChild(n); el.appendChild(s); result.push(s);
          }
        });
        return result;
      }

      // Split text chars, preserving <br> tags
      function splitChars(el) {
        var result = [];
        var nodes = Array.from(el.childNodes);
        while (el.firstChild) el.removeChild(el.firstChild);
        nodes.forEach(function (n) {
          if (n.nodeType === 3) {
            Array.from(n.textContent).forEach(function (c) {
              if (c === ' ') { el.appendChild(document.createTextNode(' ')); return; }
              var s = document.createElement('span');
              s.style.cssText = 'display:inline-block;will-change:transform,opacity';
              s.textContent = c; el.appendChild(s); result.push(s);
            });
          } else if (n.nodeType === 1 && n.nodeName === 'BR') {
            el.appendChild(n);
          }
        });
        return result;
      }

      // Bypass CSS stagger on an element
      function bypass(el) {
        if (!el) return;
        el.style.opacity = '1';
        el.style.transform = 'none';
        el.style.transition = 'none';
      }

      var heroFig = document.querySelector('#home .hero-fig');
      var h1 = document.querySelector('#home .hero-h1');
      var h2 = document.querySelector('#home .hero-h2');

      // Bypass CSS stagger on h1 and h2 — desc and btns have their own CSS/TextTyper animations
      bypass(h1); bypass(h2);

      // Hero figure: start hidden below
      if (heroFig) gsap.set(heroFig, { opacity: 0, y: 70 });

      // Split h1 and h2

      var h2Chars = h2 ? splitChars(h2) : [];

          if (h2Chars.length)
            gsap.to(h2Chars, { opacity: 1, y: 0, duration: 0.7, ease: EASE, stagger: 0.035, delay: 0.6 });
        }, 350);
      }

      // 等 loader 消失後才執行（由 hero-ready event 觸發）
      window.addEventListener('hero-ready', function () { animate(); }, { once: true });
    }

      function hexToRGB01(hex) {
        const c = hex.replace('#', '');
        return [parseInt(c.slice(0, 2), 16) / 255, parseInt(c.slice(2, 4), 16) / 255, parseInt(c.slice(4, 6), 16) / 255];

      /* ── GLSL noise (Classic Perlin 3D + value noise 2D) ── */
      const NOISE_GLSL = `
float random(in vec2 st){return fract(sin(dot(st.xy,vec2(12.9898,78.233)))*43758.5453123);}
float noise(in vec2 st){
  vec2 i=floor(st),f=fract(st);
  float a=random(i),b=random(i+vec2(1.,0.)),c=random(i+vec2(0.,1.)),d=random(i+vec2(1.,1.));
  vec2 u=f*f*(3.-2.*f);
  return mix(a,b,u.x)+(c-a)*u.y*(1.-u.x)+(d-b)*u.x*u.y;
}
vec4 permute(vec4 x){return mod(((x*34.)+1.)*x,289.);}
vec4 taylorInvSqrt(vec4 r){return 1.79284291400159-0.85373472095314*r;}
vec3 fade(vec3 t){return t*t*t*(t*(t*6.-15.)+10.);}
float cnoise(vec3 P){
  vec3 Pi0=floor(P),Pi1=Pi0+vec3(1.);
  Pi0=mod(Pi0,289.);Pi1=mod(Pi1,289.);
  vec3 Pf0=fract(P),Pf1=Pf0-vec3(1.);
  vec4 ix=vec4(Pi0.x,Pi1.x,Pi0.x,Pi1.x),iy=vec4(Pi0.yy,Pi1.yy);
  vec4 iz0=Pi0.zzzz,iz1=Pi1.zzzz;
  vec4 ixy=permute(permute(ix)+iy);
  vec4 ixy0=permute(ixy+iz0),ixy1=permute(ixy+iz1);
  vec4 gx0=ixy0/7.,gy0=fract(floor(gx0)/7.)-.5;gx0=fract(gx0);
  vec4 gz0=vec4(.5)-abs(gx0)-abs(gy0),sz0=step(gz0,vec4(0.));
  gx0-=sz0*(step(0.,gx0)-.5);gy0-=sz0*(step(0.,gy0)-.5);
  vec4 gx1=ixy1/7.,gy1=fract(floor(gx1)/7.)-.5;gx1=fract(gx1);
  vec4 gz1=vec4(.5)-abs(gx1)-abs(gy1),sz1=step(gz1,vec4(0.));
  gx1-=sz1*(step(0.,gx1)-.5);gy1-=sz1*(step(0.,gy1)-.5);
  vec3 g000=vec3(gx0.x,gy0.x,gz0.x),g100=vec3(gx0.y,gy0.y,gz0.y);
  vec3 g010=vec3(gx0.z,gy0.z,gz0.z),g110=vec3(gx0.w,gy0.w,gz0.w);
  vec3 g001=vec3(gx1.x,gy1.x,gz1.x),g101=vec3(gx1.y,gy1.y,gz1.y);
  vec3 g011=vec3(gx1.z,gy1.z,gz1.z),g111=vec3(gx1.w,gy1.w,gz1.w);
  vec4 norm0=taylorInvSqrt(vec4(dot(g000,g000),dot(g010,g010),dot(g100,g100),dot(g110,g110)));
  g000*=norm0.x;g010*=norm0.y;g100*=norm0.z;g110*=norm0.w;
  vec4 norm1=taylorInvSqrt(vec4(dot(g001,g001),dot(g011,g011),dot(g101,g101),dot(g111,g111)));
  g001*=norm1.x;g011*=norm1.y;g101*=norm1.z;g111*=norm1.w;
  float n000=dot(g000,Pf0),n100=dot(g100,vec3(Pf1.x,Pf0.yz));
  float n010=dot(g010,vec3(Pf0.x,Pf1.y,Pf0.z)),n110=dot(g110,vec3(Pf1.xy,Pf0.z));
  float n001=dot(g001,vec3(Pf0.xy,Pf1.z)),n101=dot(g101,vec3(Pf1.x,Pf0.y,Pf1.z));
  float n011=dot(g011,vec3(Pf0.x,Pf1.yz)),n111=dot(g111,Pf1);
  vec3 fade_xyz=fade(Pf0);
  vec4 n_z=mix(vec4(n000,n100,n010,n110),vec4(n001,n101,n011,n111),fade_xyz.z);
  vec2 n_yz=mix(n_z.xy,n_z.zw,fade_xyz.y);
  return 2.2*mix(n_yz.x,n_yz.y,fade_xyz.x);
}`;

      /* ── extendMaterial: ShaderMaterial inheriting from physical PBR ── */
      function extendMaterial(BaseMaterial, cfg) {
        const physical = THREE.ShaderLib.physical;
        const uniforms = THREE.UniformsUtils.clone(physical.uniforms);
        const defaults = new BaseMaterial(cfg.material || {});
        if (defaults.color) uniforms.diffuse.value = defaults.color.clone();
        if ('roughness' in defaults) uniforms.roughness.value = defaults.roughness;
        if ('metalness' in defaults) uniforms.metalness.value = defaults.metalness;
        if ('envMapIntensity' in defaults) uniforms.envMapIntensity.value = defaults.envMapIntensity;
        Object.entries(cfg.uniforms ?? {}).forEach(([k, u]) => {
          uniforms[k] = (u !== null && typeof u === 'object' && 'value' in u) ? u : { value: u };
        });
        let vert = `${cfg.header}\n${cfg.vertexHeader ?? ''}\n${physical.vertexShader}`;
        let frag = `${cfg.header}\n${cfg.fragmentHeader ?? ''}\n${physical.fragmentShader}`;
        for (const [inc, code] of Object.entries(cfg.vertex ?? {})) vert = vert.replace(inc, `${inc}\n${code}`);
        for (const [inc, code] of Object.entries(cfg.fragment ?? {})) frag = frag.replace(inc, `${inc}\n${code}`);
        return new THREE.ShaderMaterial({
          defines: { ...(physical.defines ?? {}) },
          uniforms, vertexShader: vert, fragmentShader: frag,
          lights: true, fog: !!cfg.material?.fog,
        });
      }

      /* ── Stacked planes geometry (n vertical planes centred at origin) ── */
      function createBeamGeometry(n, width, height, heightSegments) {
        const geo = new THREE.BufferGeometry();
        const spacing = 0;
        const nV = n * (heightSegments + 1) * 2;
        const nF = n * heightSegments * 2;
        const pos = new Float32Array(nV * 3);
        const idx = new Uint32Array(nF * 3);
        const uvs = new Float32Array(nV * 2);
        let vi = 0, ii = 0, ui = 0;
        const totalW = n * width + (n - 1) * spacing;
        const xBase = -totalW / 2;
        for (let i = 0; i < n; i++) {
          const xOff = xBase + i * (width + spacing);
          const uvXOff = Math.random() * 300;
          const uvYOff = Math.random() * 300;
          for (let j = 0; j <= heightSegments; j++) {
            const y = height * (j / heightSegments - 0.5);
            pos.set([xOff, y, 0, xOff + width, y, 0], vi * 3);
            const uvY = j / heightSegments;
            uvs.set([uvXOff, uvY + uvYOff, uvXOff + 1, uvY + uvYOff], ui);
            if (j < heightSegments) {
              const a = vi, b = vi + 1, c = vi + 2, d = vi + 3;
              idx.set([a, b, c, c, b, d], ii); ii += 6;
            }
            vi += 2; ui += 4;
          }
        }
        geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
        geo.setAttribute('uv', new THREE.BufferAttribute(uvs, 2));
        geo.setIndex(new THREE.BufferAttribute(idx, 1));
        geo.computeVertexNormals();
        return geo;
      }

    class TextTyper {
      constructor(textEl, cursorEl, texts, opts = {}) {
        this.textEl = textEl;
        this.cursorEl = cursorEl;
        this.texts = texts;
        this.typingSpeed = opts.typingSpeed || 75;
        this.deletingSpeed = opts.deletingSpeed || 40;
        this.pauseDuration = opts.pauseDuration || 1500;
        this.idx = 0;
        this.charIdx = 0;
        this.deleting = false;
        this.current = '';
        this.timer = null;
        this._startCursor();
        this._tick();
      }
      _startCursor() {
        if (this.cursorEl && window.gsap) {
          gsap.killTweensOf(this.cursorEl);
          gsap.set(this.cursorEl, { opacity: 1 });
          gsap.to(this.cursorEl, { opacity: 0, duration: 0.5, repeat: -1, yoyo: true, ease: 'power2.inOut' });
        }
      }
      _tick() {
        const word = this.texts[this.idx];
        if (this.deleting) {
          if (this.current === '') {
            this.deleting = false;
            this.idx = (this.idx + 1) % this.texts.length;
            this.charIdx = 0;
            this.timer = setTimeout(() => this._tick(), 300);
          } else {
            this.current = this.current.slice(0, -1);
            this.textEl.textContent = this.current;
            this.timer = setTimeout(() => this._tick(), this.deletingSpeed);
          }
        } else {
          if (this.charIdx < word.length) {
            this.current += word[this.charIdx++];
            this.textEl.textContent = this.current;
            this.timer = setTimeout(() => this._tick(), this.typingSpeed);
          } else {
            this.timer = setTimeout(() => { this.deleting = true; this._tick(); }, this.pauseDuration);
          }
        }
      }
      stop() {
        clearTimeout(this.timer);
        if (this.cursorEl && window.gsap) gsap.killTweensOf(this.cursorEl);
      }
      restart(newTexts) {
        this.stop();
        this.texts = newTexts;
        this.idx = 0; this.charIdx = 0; this.deleting = false; this.current = '';
        this.textEl.textContent = '';
        this._startCursor();
        this._tick();
      }
    }

  <script>
    (function () {
      var fab = document.getElementById('chat-fab');
      var panel = document.getElementById('chat-panel');
      var messages = document.getElementById('chat-messages');
      var input = document.getElementById('chat-input');
      var sendBtn = document.getElementById('chat-send');

      fab.addEventListener('click', function () {
        panel.classList.toggle('open');
        if (panel.classList.contains('open')) input.focus();
      });

      document.querySelectorAll('.chat-quick-btn').forEach(function (btn) {
        btn.addEventListener('click', function () {
          sendQuestion(btn.getAttribute('data-q'));
        });
      });

      input.addEventListener('keydown', function (e) {
        if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendQuestion(input.value); }
      });
      sendBtn.addEventListener('click', function () { sendQuestion(input.value); });

      function addMsg(text, role) {
        var el = document.createElement('div');
        el.className = 'chat-msg ' + role;
        el.textContent = text;
        messages.appendChild(el);
        messages.scrollTop = messages.scrollHeight;
        return el;
      }

      function sendQuestion(q) {
        q = (q || '').trim();
        if (!q) return;
        input.value = '';
        sendBtn.disabled = true;

        addMsg(q, 'user');
        var loading = addMsg('…', 'bot loading');

        fetch('/api/chat', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ question: q }),
        })
          .then(function (r) { return r.json(); })
          .then(function (data) {
            loading.textContent = data.reply || data.error || '無法取得回覆，請直接聯繫 Tim。';
            loading.classList.remove('loading');
          })
          .catch(function () {
            loading.textContent = '連線失敗，請直接透過聯絡表單聯繫 Tim。';
            loading.classList.remove('loading');
          })
          .finally(function () {
            sendBtn.disabled = false;
            input.focus();
          });
      }
    })();
  </script>
